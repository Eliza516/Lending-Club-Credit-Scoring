\section{Problem Statement}

\begin{frame}{Problem Statement}
    \textbf{Challenge}: Rank loan applicants by default risk using only information known when the loan is approved

    \vspace{0.4cm}
    \textbf{Key Issues}:
    \begin{itemize}
        \item \textbf{Data leakage}: many of the 151 columns are created after the loan starts
        \item \textbf{Temporal drift}: default rate rises from 13.7\% (train) to 14.8\% (test)
        \item \textbf{Right-censoring}: recent loans have no final outcome yet
        \item \textbf{Incumbent model}: Lending Club's own \code{sub\_grade} (A1--G5) is already a risk score
    \end{itemize}

    \vspace{0.4cm}
    \textbf{Solution}: A leakage-controlled, out-of-time pipeline compared against the \code{sub\_grade} baseline
\end{frame}

\begin{frame}{Task Definition and Metrics}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \textbf{Binary classification:} will a new loan be charged off?
            \vspace{0.2cm}
            \begin{itemize}
                \item \textbf{Input}: data known at approval (amount, income, job, credit history)
                \item \textbf{Model}: scoring model trained on old loans
                \item \textbf{Output}: $P(\text{default})\in[0,1]$, used to rank risk
                \item \textbf{Label}: $0$ = Fully Paid, $1$ = Charged Off
            \end{itemize}
            \textcolor{red}{Main rule}: no data created after approval
        \end{column}
        \begin{column}{0.5\textwidth}
            \textbf{Metrics (ranking quality):}
            \begin{itemize}
                \item \textbf{ROC-AUC}: chance a bad loan scores above a good loan
                \item \textbf{Gini} $=2\times\mathrm{AUC}-1$: standard score in credit risk
                \item \textbf{KS}: max gap between bad and good cumulative rates, $\max(\mathrm{TPR}-\mathrm{FPR})$
            \end{itemize}
            No probability cut-off is needed: all three measure ranking.
        \end{column}
    \end{columns}
\end{frame}

\section{Existing Methodologies}

\begin{frame}{Related Work Overview}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \textbf{Studies on Lending Club data:}
            \begin{itemize}
                \item Serrano-Cinca et al.: grade is the most predictive factor; logistic regression \cite{serrano2015}
                \item Malekipirbazari \& Aksakalli: random forests beat logistic regression, FICO and LC grades \cite{malekipirbazari2015}
                \item \textcolor{red}{Limitation}: different data, filters and splits; not comparable
            \end{itemize}
        \end{column}
        \begin{column}{0.5\textwidth}
            \textbf{Evaluation over time:}
            \begin{itemize}
                \item Qian et al.: mixing periods leaks the future and inflates results \cite{qian2021}
                \item Open-source projects: only some use out-of-time splits and the platform grade as benchmark \cite{gh_arturo,gh_vaibhav}
                \item \textcolor{red}{Limitation}: leakage and time control rarely verified
            \end{itemize}
        \end{column}
    \end{columns}

    \vspace{0.5cm}
    \textbf{Our Contribution}: A reproducible study with an out-of-time split, a \code{sub\_grade} baseline with bootstrap intervals, and an ablation without \code{sub\_grade}
\end{frame}

\begin{frame}{Data Filtering and Out-of-Time Split}
    \textbf{Train on old loans, test on newer loans: 70 / 15 / 15 by date}
    \vspace{0.2cm}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \textbf{Filter funnel (rows):}\par\vspace{0.1cm}
            \hb{Raw data}{2260}{2260}{gray}{2,260,701}
            \hb{Labelled}{1345}{2260}{blue}{1,345,310}
            \hb{Issued from 2012}{1305}{2260}{blue}{1,305,524}
            \hb{Matured}{634}{2260}{blue}{633,767}
            \hb{36-month term}{589}{2260}{bargreen}{589,488}
        \end{column}
        \begin{column}{0.5\textwidth}
            \textbf{Three sets:}
            \begin{itemize}
                \item \textbf{Train} 408,188 (69.2\%): 2012-01 $\rightarrow$ 2015-05, default 13.7\%
                \item \textbf{Val} 92,631: 2015-06 $\rightarrow$ 09, default 14.7\%
                \item \textbf{Test} 88,669: 2015-10 $\rightarrow$ 12, default 14.8\%
            \end{itemize}
        \end{column}
    \end{columns}
    \vspace{0.2cm}
    \begin{itemize}
        \item \textbf{Matured}: issue month + term $\le$ 2018-12 (99.6\% of matured loans are labelled vs 42.4\%)
        \item No month in two sets; default rate drifts up $+1.1$ points
    \end{itemize}
\end{frame}

\begin{frame}{Data Cleaning and Leakage Prevention}
    \textbf{Each rule checks one row; the same rule runs on train, val and test}
    \begin{table}
        \centering
        \scriptsize
        \begin{tabular}{rlll}
            \hline
            \textbf{\#} & \textbf{Logic rule (check)} & \textbf{Why} & \textbf{Handling} \\
            \hline
            1 & \code{annual\_inc} $\le 0$ & Income must be positive & Set to missing \\
            2 & \code{loan\_amnt} $\le 0$ & Amount must be positive & Set to missing \\
            3 & \code{dti} $<0$ or $>100$ & DTI must be 0--100 & Set to missing \\
            4 & \code{revol\_util} $>100$ & Possible; extremes are noise & Cap at 102 (train p99.9) \\
            5 & \code{fico\_low} $>$ \code{fico\_high} & Low must not pass high & Set both to missing \\
            6 & \code{earliest\_cr\_line} $>$ \code{issue\_d} & Credit before the loan & Set to missing \\
            \hline
        \end{tabular}
    \end{table}
    \textbf{Anti-leakage rules:}
    \begin{enumerate}
        \item Split first, transform later
        \item Median, mean, std, caps: \textbf{train only}
        \item Pipeline \code{fit} on train; val and test only \code{transform}
    \end{enumerate}
\end{frame}

\section{Conclusion}

\begin{frame}{Conclusions}
    \textbf{Summary:}
    \begin{itemize}
        \item Logistic L2 and HistGB beat the \code{sub\_grade} baseline out of time (test AUC 0.7039 and 0.7055 vs 0.6818) and cannot be told apart
        \item Without \code{sub\_grade}, Logistic still beats the baseline (0.6879), and FICO and income coefficients about double
        \item A compact 12-field scorecard is below the baseline in AUC (0.6730) but separates risk bands from 30.2\% to 3.3\% default
    \end{itemize}

    \vspace{0.3cm}
    \textbf{Key Technical Contributions:}
    \begin{enumerate}
        \item \textbf{Leakage control}: application-time whitelist checked by single-column AUC
        \item \textbf{Out-of-time protocol}: matured 36-month loans, chronological 70/15/15 split, pipeline fitted on train only
        \item \textbf{Honest evaluation}: bootstrap intervals, single test use, negative results reported
    \end{enumerate}
\end{frame}

\begin{frame}{Limitations and Future Work}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \textbf{Limitations:}
            \begin{itemize}\small
                \item 36-month loans of 2012--2015; 3-month test window
                \item Random Forest and HistGB grids hit their edges
                \item Bootstrap CI covers validation sampling only
                \item Val AUC $>$ train AUC for the baseline and Logistic: not explained
                \item No calibration or cost analysis; scorecard constants are conventions
            \end{itemize}
        \end{column}
        \begin{column}{0.5\textwidth}
            \textbf{Research Directions:}
            \begin{enumerate}\small
                \item Wider hyperparameter search
                \item Rolling-origin evaluation and dataset-shift tests \cite{qian2021}
                \item Missing-indicator treatment of \code{mths\_since\_...} columns
                \item Calibration and cost-sensitive thresholds
                \item 60-month loans via survival analysis
            \end{enumerate}
        \end{column}
    \end{columns}
\end{frame}
