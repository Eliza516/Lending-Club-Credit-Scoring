\section{Problem Statement}

\begin{frame}{Problem Statement}
    \textbf{Challenge}: Blockchain fraud detection is critical for maintaining security and integrity
    
    \vspace{0.5cm}
    \textbf{Key Issues}:
    \begin{itemize}
        \item Malicious actors exploit pseudonymous blockchain transactions
        \item Limited attention to feature selection in existing approaches
        \item Power-law distribution of account transaction frequencies
        \item Topological noise from fraudulent-legitimate account interactions
        \item Insufficient discriminative power in current temporal GNNs
    \end{itemize}
    
    \vspace{0.5cm}
    \textbf{Solution}: Effective framework combining feature selection + temporal graph modeling
\end{frame}

\section{Existing Methodologies}

\begin{frame}{Related Work Overview}

                \item Handle high-dimensional data
                \item Examples: Eth-PSD, XRAD, Block Hunter
                \item \textcolor{red}{Limitation}: Manual feature engineering
            \end{itemize}
        \end{column}
        \begin{column}{0.5\textwidth}

            \begin{itemize}
                \item Static GNNs: GCN, GAT
                \item Temporal GNNs: PDTGA, TTAGN, DiT-SGCR
                \item \textcolor{red}{Limitation}: Lack sufficient discriminative power
            \end{itemize}
        \end{column}
    \end{columns}
    
    \vspace{0.5cm}

\section{Proposed Method}

\begin{frame}{Proposed Method}
    \begin{figure}
        \centering
        % Trim parameters: <Left> <Bottom> <Right> <Top>
        \includegraphics[width=0.95\textwidth, trim=0cm 0cm 8cm 0cm, clip]{figs/STGraph_FS.drawio.pdf}
        \caption{The overall framework of STGraph-FS with four key phases: (1) Graph Initialization, (2) Spearman-Guided Directed Temporal Aggregation, (3) Soft Clustering, and (4) Binary Classification}
        \label{fig:framework}
    \end{figure}
\end{frame}

\begin{frame}{STGraph-FS Architecture}
    \textbf{Key Components:}
    \begin{enumerate}
        \item \textbf{Graph Initialization}: Construct directed temporal graph from Ethereum data
        \item \textbf{Spearman-Guided Directed Temporal Aggregation (SG-DTA)}:
        \begin{itemize}
            \item Filter discriminative features using Spearman correlation
            \item Aggregate directed neighbor information and temporal dynamics
        \end{itemize}
        \item \textbf{Soft Clustering}: 
        \begin{itemize}
            \item Capture global structural patterns
            \item Representation reduction and clustering optimization
        \end{itemize}
        \item \textbf{Binary Classification}: XGBoost for final fraud detection
    \end{enumerate}
\end{frame}

\begin{frame}{Graph Initialization}
    \textbf{Directed Temporal Graph Construction:}
    \begin{itemize}
        \item $G = (V, E, T)$ where:
        \begin{itemize}
            \item $V$: Set of all accounts
            \item $E \subseteq V \times V \times \mathbb{R}$: Directed transactions with timestamps
            \item $T = \{t_i \mid i = 1, \dots, K\}$: All recorded timestamps
        \end{itemize}
        \item Each node $u \in V$ records chronological interactions
        \item $N_{in}(u, t_i)$ and $N_{out}(u, t_i)$: Incoming and outgoing neighbors
    \end{itemize}
    
    \vspace{0.5cm}
    \textbf{Objective:}
    Learn node embeddings $H \in \mathbb{R}^{|V| \times d}$ that encode both structural connections and temporal dynamics with discriminative power.
\end{frame}

\begin{frame}{Spearman-Guided Feature Selection}
    \textbf{Feature Selection Process:}
    \begin{enumerate}
        \item Each node $u$ has feature vector $\mathbf{f}_u = [f_u^{1}, f_u^{2}, \dots, f_u^{M}]$
        \item Compute Spearman's rank correlation coefficient:
        \begin{align}
        \rho_m = 1 - \frac{6 \sum_{u=1}^{|V|} (r(f_u^{m}) - r(y_u))^2}{|V|(|V|^2 - 1)}
        \end{align}
        \item Select features where $|\rho_m| \ge \tau$ (threshold)
        \item Resulting selected features: $\mathbf{f}^*_u = [f^{1}_u, f^{2}_u, \dots, f^{P}_u]$, $P < M$
    \end{enumerate}
    
    
\vspace{0.3cm}
\end{frame}

\begin{frame}{Directed Neighbors Aggregation}
    \textbf{Directed Aggregation:}
    \begin{align}
        \mathbf{x}_{\text{in}}(u, t_i) &= \sum_{v \in N_{\text{in}}(u, t_i)} \mathbf{e}_v \\
        \mathbf{x}_{\text{out}}(u, t_i) &= \sum_{v \in N_{\text{out}}(u, t_i)} \mathbf{e}_v
    \end{align}
    
    \textbf{Normalize concatenated representations:}
    \begin{align}
        \mathbf{x}(u, t_i) = \frac{[\mathbf{x}_{\text{in}}(u, t_i), \mathbf{x}_{\text{out}}(u, t_i)]}{\|[\mathbf{x}_{\text{in}}(u, t_i), \mathbf{x}_{\text{out}}(u, t_i)]\|_2 + \epsilon}
    \end{align}
    
    \textbf{Directed Neighbor Aggregation:}
    \begin{align}
        \mathbf{m}_u = \sum_{i=1}^{T_u} \mathbf{x}(u, t_i)
    \end{align}
\end{frame}

\begin{frame}{Temporal Aggregation}
    \textbf{Temporal Evolution with Exponential Decay:}
    \begin{align}
    \mathbf{s}(u,t_i) &= \exp\!\left(-\frac{t_i - t_{i-1}}{\alpha}\right) \cdot (\mathbf{x}(u, t_{i-1}) + \mathbf{s}(u, t_{i-1})) \\
    \mathbf{s}'(u,t_i) &= \frac{\mathbf{s}(u,t_i)}{\|\mathbf{s}(u,t_i)\|_2 + \epsilon}
    \end{align}
    
    \textbf{Structural-Temporal Fusion:}
    \begin{align}
    \mathbf{a}(u, t_i) &= \mathbf{x}(u, t_i) \otimes \mathbf{s}'(u, t_i) \\
    \mathbf{D}_u &= \sum_{i=1}^{T_u} \mathbf{a}(u, t_i) \in \mathbb{R}^{C \times C}
    \end{align}
    
    where:
    \begin{itemize}
        \item $\alpha$: Decay factor controlling temporal sensitivity
        \item $\mathbf{s}'(u,t_i)$: Normalized temporal state vector
        \item $\mathbf{a}(u, t_i)$: Outer product capturing interaction patterns
        \item $\mathbf{D}_u$: Final interaction matrix for node $u$
    \end{itemize}
\end{frame}

\begin{frame}{Final Embedding Generation}
    \textbf{Complete Node Embedding:}
    \begin{align}
    \mathbf{h}_u = [\operatorname{R}(\mathbf{D}_u), \, \mathbf{m}_u, \, \mathbf{f}_u^{*}]
    \end{align}
    
    where:
    \begin{itemize}
        \item $\operatorname{R}(\mathbf{D}_u)$: Flattened structural-temporal interaction matrix
        \item $\mathbf{m}_u$: Temporal summary embedding  
        \item $\mathbf{f}_u^{*}$: Selected discriminative features
    \end{itemize}
    
    \vspace{0.5cm}
\end{frame}


\begin{frame}{Soft Clustering Module}
    \textbf{Clustering Process:}
    \begin{enumerate}
        \item \textbf{Centroid Init}: K-means++ for centroids $\boldsymbol{\mu}_c \in \mathbb{R}^{4C^2 + 2C + P}$
        \item \textbf{Cosine Similarity}: 
        \begin{align}
        \text{sim}_{\cos}(u, c) = \frac{\mathbf{h}_u^{\top} \boldsymbol{\mu}_c}{\|\mathbf{h}_u\|_2 \, \|\boldsymbol{\mu}_c\|_2}
        \end{align}
        \item \textbf{Soft Assignment}: 
        \begin{align}
        w_{u,c} = \frac{\exp(\beta \cdot \text{sim}_{\cos}(u, c))}{\sum_{c'=1}^{C} \exp(\beta \cdot \text{sim}_{\cos}(u, c'))}
        \end{align}
        \item \textbf{Centroid Update}: 
        \begin{align}
        \boldsymbol{\mu}_c = \frac{\sum_{u \in V} w_{u,c} \mathbf{h}_u}{\sum_{u \in V} w_{u,c} + \epsilon}
        \end{align}
    \end{enumerate}
\end{frame}

\begin{frame}{Representation Reduction \& Optimization}
    \textbf{Representation Reduction:}
    \begin{align}
    \text{dist}_{u,c} &= 1 - \text{sim}_{\cos}(u, c) \\
    \bar{z}_{u,c} &= \frac{\max_c(\text{dist}_{u,c}) - \text{dist}_{u,c}}{\max_c(\text{dist}_{u,c}) - \min_c(\text{dist}_{u,c}) + \epsilon} \\
    z_{u,c} &= \frac{\bar{z}_{u,c}}{\sum_{c'=1}^{C} \bar{z}_{u,c'} + \epsilon}
    \end{align}
    
    \textbf{Clustering Optimization:}
    \begin{align}
    \min_{\mathbf{Z}'} &\; \text{tr}(\mathbf{Z}'^{\top} \mathbf{L} \mathbf{Z}') + \gamma \sum_{c=1}^C \text{tr}(\mathbf{Z}'^{\top} \mathbf{L}_c \mathbf{Z}') + \phi \|\mathbf{Z}' - \mathbf{Q}\|_F^2
    \end{align}
    
    where $\mathbf{Z} \in \mathbb{R}^{|V| \times C}$ captures global topology relative to cluster centroids.
\end{frame}

\section{Conclusion}

\begin{frame}{Conclusions}
    \textbf{Summary:}
    \begin{itemize}
        \item We present \textbf{STGraph-FS}, a supervised, graph-based framework for Ethereum fraud detection
        \item Leverages complementary strengths of optimal feature selection and directed temporal graph modeling
        \item Demonstrates effectiveness in robustly handling heavily imbalanced data distributions
    \end{itemize}
    
    \vspace{0.5cm}
    \textbf{Key Technical Contributions:}
    \begin{enumerate}
        \item \textbf{SG-DTA Module}: Spearman-guided mechanism to identify and prioritize discriminative features, mitigating noise in high-dimensional transaction data
        \item \textbf{Soft Clustering}: Seamlessly incorporates structural information into embeddings with clustering optimization
        \item \textbf{Superior Performance}: Outperforms state-of-the-art baselines on real-world MulDiGraph dataset
    \end{enumerate}
\end{frame}

\begin{frame}{Future Work}
    \textbf{Research Directions:}
    \begin{enumerate}
        \item \textbf{Real-time Systems}: Extend framework to real-time fraud detection systems
        \item \textbf{Cross-chain Networks}: Explore applicability to cross-chain transaction networks with complex inter-chain relationships  
        \item \textbf{Advanced Security}: Address greater security challenges in multi-blockchain environments
        \item \textbf{Scalability}: Optimize for larger and more complex blockchain networks
    \end{enumerate}
    
    \vspace{0.5cm}
\end{frame}
