%% ================= MỞ ĐẦU =================
\begin{frame}{Mở đầu: Đặt vấn đề}
\begin{columns}[T]
\column{0.58\textwidth}
\textbf{Lending Club} là nền tảng cho vay ngang hàng (peer-to-peer) của Mỹ. Mỗi khoản vay kết thúc ở một trong hai trạng thái: \alert{Charged Off} (vỡ nợ, nhãn 1) hoặc \alert{Fully Paid} (trả đủ, nhãn 0).

\medskip
\textbf{Câu hỏi:} nếu chỉ dùng thông tin biết được lúc xét duyệt, ta xếp hạng rủi ro người vay tốt đến đâu, và mô hình có hơn \code{sub\_grade} (A1--G5) của chính nền tảng không?

\medskip
\textbf{Thách thức}
\begin{itemize}
\item \textbf{Rò rỉ dữ liệu}: nhiều cột trong 151 cột chỉ có sau khi vay
\item \textbf{Thời gian}: tỷ lệ vỡ nợ trôi từ 13.7\% lên 14.8\%
\item \textbf{Kiểm duyệt}: khoản vay mới chưa có kết cục
\item \textbf{Đối thủ có sẵn}: \code{sub\_grade} đã là một mô hình rủi ro
\end{itemize}
\column{0.4\textwidth}
\begin{block}{Bài toán}
\textbf{Đầu vào}: thông tin người vay và hồ sơ tín dụng lúc xét duyệt\\
\textbf{Đầu ra}: điểm rủi ro $p\in[0,1]$ (xác suất vỡ nợ) để xếp hạng\\
\textbf{Nhãn}: Charged Off $=1$, Fully Paid $=0$
\end{block}
\begin{alertblock}{Nguyên tắc số một}
Không dùng dữ liệu phát sinh sau khi duyệt vay.
\end{alertblock}
\end{columns}
\end{frame}

\begin{frame}{Mở đầu: Mục tiêu và đóng góp}
\textbf{Mục tiêu:} ước lượng trung thực thông tin lúc xét duyệt bổ sung được bao nhiêu so với \code{sub\_grade} khi xếp hạng rủi ro vỡ nợ.

\medskip
\begin{block}{Đóng góp chính}
\begin{itemize}
\item Quy trình chống rò rỉ: chỉ dùng cột biết lúc xét duyệt, có bằng chứng bằng AUC từng cột
\item Chia Out-of-Time 70/15/15 trên khoản vay 36 tháng đã đáo hạn
\item Pipeline chỉ học trên train (điền trung vị, chuẩn hóa, one-hot)
\item So sánh Baseline, Logistic L2, Decision Tree, Random Forest, HistGB bằng ROC-AUC, Gini, KS và bootstrap
\item Hai thí nghiệm mở rộng: bỏ \code{sub\_grade}, thẻ điểm 12 biến
\item Báo cáo trung thực cả kết quả không như kỳ vọng
\end{itemize}
\end{block}
\end{frame}

%% ================= PHỤ LỤC =================
\begin{frame}{}
\slidetitle{PHỤ LỤC}{Các thước đo đánh giá}
\begin{columns}[T]
\column{0.5\textwidth}
\textbf{ROC-AUC}: xác suất mô hình cho khoản vỡ nợ điểm cao hơn khoản trả đủ\\[0.2cm]
\textbf{Gini} $=2\times$ AUC $-1$: thước đo chuẩn trong rủi ro tín dụng\\[0.2cm]
\textbf{KS}: khoảng cách lớn nhất giữa đường tích lũy của nhóm vỡ nợ và nhóm trả đủ, $\max(\mathrm{TPR}-\mathrm{FPR})$\\[0.3cm]
Không cần chọn ngưỡng xác suất: cả ba thước đo đều đo khả năng xếp hạng. Cùng một hàm dùng cho mọi mô hình và mọi tập dữ liệu.
\column{0.48\textwidth}
\footnotesize
\begin{block}{NB03 cell 5}
\texttt{def calculate\_metrics(y\_true, proba):}\\
\texttt{\ \ auc = roc\_auc\_score(y\_true, proba)}\\
\texttt{\ \ gini = 2 * auc - 1}\\
\texttt{\ \ fpr, tpr, \_ = roc\_curve(y\_true, proba)}\\
\texttt{\ \ ks = np.max(tpr - fpr)}\\
\texttt{\ \ return auc, gini, ks}
\end{block}
\end{columns}
\end{frame}
