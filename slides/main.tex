\begin{frame}[fragile]{Evaluation Metrics}
  \begin{columns}
    \begin{column}{0.5\textwidth}
      \begin{itemize}
        \item \textbf{ROC-AUC}: how well the model ranks bad loans above good loans
        \item \textbf{Gini} $=2\times\mathrm{AUC}-1$
        \item \textbf{KS}: max gap between bad and good cumulative rates
      \end{itemize}
      No probability cut-off is needed: all three metrics measure ranking quality. The same function is used for every model and every data set.
    \end{column}

      \textbf{\scriptsize NB03 $\cdot$ cell 5}
\begin{lstlisting}
def calculate_metrics(y_true, proba):
    auc = roc_auc_score(y_true, proba)
    gini = 2 * auc - 1
    fpr, tpr, _ = roc_curve(y_true, proba)
    ks = np.max(tpr - fpr)
    return auc, gini, ks
\end{lstlisting}
    \end{column}

\begin{frame}[fragile]{Step 3 Details: time\_split Function}
  \scriptsize 1. Sort by \code{issue\_date}, then \code{id} \ $\cdot$ \ 2. Find cut dates at 70\% and 85\% of rows \ $\cdot$ \ 3. Split by date: a whole month always stays in one set\\[0.1cm]
  \textbf{NB01 $\cdot$ cell 90}
\begin{lstlisting}
def time_split(df, train_share=0.70, val_share=0.15):
    df = df.sort_values(["issue_date", "id"]).reset_index(drop=True)
    cut_val = df["issue_date"].iloc[int(len(df) * train_share)]
    cut_test = df["issue_date"].iloc[int(len(df) * (train_share + val_share))]
    train = df[df["issue_date"] < cut_val]
    val = df[(df["issue_date"] >= cut_val) & (df["issue_date"] < cut_test)]
    test = df[df["issue_date"] >= cut_test]
    return train, val, test, cut_val, cut_test
# Output: Validation starts: 2015-06-01 / Test starts: 2015-10-01
\end{lstlisting}
\end{frame}

def fix_invalid_values(df):
    df = df.copy()
    df.loc[df["annual_inc"] <= 0, "annual_inc"] = np.nan
    df.loc[df["loan_amnt"] <= 0, "loan_amnt"] = np.nan
    df.loc[(df["dti"] < 0) | (df["dti"] > DTI_MAX), "dti"] = np.nan
    df["revol_util"] = df["revol_util"].clip(upper=REVOL_UTIL_CAP)
    bad_fico = df["fico_range_low"] > df["fico_range_high"]
    df.loc[bad_fico, ["fico_range_low", "fico_range_high"]] = np.nan
    first_credit = parse_month(df["earliest_cr_line"])
    df.loc[first_credit > df["issue_date"], "earliest_cr_line"] = np.nan
    return df

    if column in identifier_cols:         return "identifier"
    if column in empty_cols:              return "empty in train"
    if timing[column] in ("No", "Uncertain"): return "after loan / unclear timing"
    if column in joint_columns:           return "joint application"
    if column in many_values:             return "text with too many values"
    if p["top_value_share"] >= TOP_SHARE_CONSTANT: return "near constant"
    if p["missing_share"] > MAX_MISSING:  return "too many missing in train"

def single_column_auc(df, column):
    series = df[column]
    if pd.api.types.is_numeric_dtype(series):
        values = series.fillna(series.median())
    elif series.nunique() <= 100:
        values = df.groupby(series.fillna("(missing)"))["default"].transform("mean")
    else:
        values = series.notna().astype(int)
    auc = roc_auc_score(df["default"], values)
    return max(auc, 1 - auc)

def high_corr_pairs(corr_matrix, threshold):
    upper = corr_matrix.where(np.triu(np.ones(corr_matrix.shape, dtype=bool), k=1)).stack().dropna()
    return upper[upper.abs() >= threshold].sort_values(key=abs, ascending=False)

\begin{frame}{Step 6 Details: Why Median and Scaling?}
  \begin{itemize}
    \item \textbf{31} numeric columns have missing values in train
    \item Most are right-skewed, so the mean is pulled up by outliers (mean is up to \textbf{+138\%} above the median)
    \item Example: \code{bc\_open\_to\_buy} mean is 138.2\% above median; \code{avg\_cur\_bal} 100.7\%
    \item Median keeps the center of typical borrowers
    \item Scales differ by \textbf{$>100{,}000\times$}: from 0.3 (\code{delinq\_2yrs}) to 127,000 USD (\code{tot\_cur\_bal})
    \item StandardScaler ($\mu=0$, $\sigma=1$) helps the L2 penalty treat all features fairly
  \end{itemize}
\end{frame}
